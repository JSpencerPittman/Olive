# Comments
COMMENT_MULTI_START := / <:asterisk:>
COMMENT_MULTI_END := <:asterisk:> /
SPECIAL-RULE := PURGE-COMMENTS

# Identifiers
IDENTIFIER := NONDIGIT ( ( NONDIGIT DIGIT _ ) | ) *
SPECIAL-RULE := CONSOLIDATE-IDENTIFIERS

# Numbers
INTEGER := ( DIGIT ) +
FLOAT := DIGIT . ( DIGIT ) *

# Keywords
SPECIAL-RULE := RESOLVE-KEYWORDS

# Macros
DEFINE_START := # KEYWORD__DEFINE ( WHITESPACE ) * IDENTIFIER
SPECIAL-RULE := CONSOLIDATE-DEFINE-CONTENTS
SPECIAL-RULE := PURGE-WHITESPACE
MACRO__DEFINE := DEFINE_START DEFINE_CONSOLIDATED

# Array
SPECIAL-RULE := CONSOLIDATE-ARRAY-BRACKETS

# Variable
VARIABLE__INCOMPLETE := ( KEYWORD__STATIC ) ? ( KEYWORD__CONST ) ? ( KEYWORD__STRUCT ) ? ( KEYWORD__UNSIGNED ) ? IDENTIFIER ( <:asterisk:> ) * IDENTIFIER ( ARRAY_BRACKET_CONSOLIDATED ) ?
VARIABLE__COMMA_TERM := VARIABLE__INCOMPLETE ,
VARIABLE__SEMICOLON_TERM := VARIABLE__INCOMPLETE ;
VARIABLE__COMMA_DEC := VARIABLE__COMMA_TERM ( ( <:asterisk:> ) * IDENTIFIER ( ARRAY_BRACKET_CONSOLIDATED ) ? , ) * ( <:asterisk:> ) * IDENTIFIER ( ARRAY_BRACKET_CONSOLIDATED ) ? ;
VARIABLE__DEC := ( VARIABLE__SEMICOLON_TERM VARIABLE__COMMA_DEC ) |
VARIABLE__LIST := ( VARIABLE__COMMA_TERM ) * VARIABLE__INCOMPLETE
